# Search UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a frontend search experience (debounced navbar search box + `/search` results page) that consumes the existing `GET /search/{arg}` backend endpoint.

**Architecture:** Pure frontend addition, no backend changes. `SearchComponent` mirrors the existing `PostListComponent` pattern (query-param-driven, paginated list of `PostSummary`). `AppComponent`'s navbar gets a debounced `Subject`-based search input that navigates to `/search?q=...`.

**Tech Stack:** Angular 22 (standalone components), RxJS (`Subject`, `debounceTime`, `distinctUntilChanged`), Angular Router (`queryParamMap`), `HttpClient`.

## Global Constraints

- Standalone components only, no NgModules (matches existing codebase).
- File/folder naming: kebab-case folders (`pages/search/search.component.ts`), class name `SearchComponent`.
- Route paths are plain lowercase English words, not kebab-case segments (existing convention: `/posts`, `/login`) — this feature's path is `/search`.
- No frontend test infrastructure exists in this repo (no test runner configured in `devblog-ui/package.json`) — do not invent test steps; verify each task via `ng build` (type-check/compile) and manual browser testing via `npm start`, per this repo's known-debt note in CLAUDE.md.
- Follow existing `PostListComponent` conventions exactly: `ChangeDetectorRef` + `detectChanges()` after async updates, `CommonModule` + `RouterLink` imports, Bootstrap utility classes (`container`, `list-group`, `btn btn-sm`, etc.) in templates.

    const params = new HttpParams().set('page', page).set('pageSize', pageSize);
    return this.http.get<PagedResult<PostSummary>>(
      `${environment.apiUrl}/search/${encodeURIComponent(term)}`,

    );
  }

import { ChangeDetectorRef, Component, inject, OnInit } from '@angular/core';
import { ActivatedRoute, Router, RouterLink } from '@angular/router';
import { CommonModule } from '@angular/common';
import { PostService, PostSummary } from '../../services/post.service';
import { AuthService } from '../../services/auth.service';

@Component({
  selector: 'app-search',
  standalone: true,
  imports: [RouterLink, CommonModule],
  templateUrl: './search.component.html',
  styleUrl: './search.component.scss'
})
export class SearchComponent implements OnInit {
  private postService = inject(PostService);
  private authService = inject(AuthService);
  private cdr = inject(ChangeDetectorRef);
  private route = inject(ActivatedRoute);
  private router = inject(Router);

  term = '';
  posts: PostSummary[] = [];
  currentPage = 1;
  pageSize = 20;
  totalPages = 0;
  totalCount = 0;
  searched = false;

  ngOnInit() {
    this.route.queryParamMap.subscribe((params) => {
      this.term = params.get('q') ?? '';
      const page = Number(params.get('page')) || 1;

      if (!this.term.trim()) {
        this.posts = [];
        this.totalPages = 0;
        this.totalCount = 0;
        this.searched = false;
        this.cdr.detectChanges();
        return;
      }

      this.loadResults(this.term, page);
    });
  }

  goToPage(page: number) {
    if (page < 1 || page > this.totalPages || page === this.currentPage) {
      return;
    }
    this.router.navigate([], { queryParams: { q: this.term, page } });
  }

  get pageNumbers(): number[] {
    return Array.from({ length: this.totalPages }, (_, i) => i + 1);
  }

  toggleLike(post: PostSummary) {
    if (!this.authService.isLoggedIn()) {
      this.router.navigate(['/login']);
      return;
    }
    this.postService.toggleLike(post.slug).subscribe((result) => {
      post.likedByCurrentUser = result.liked;
      post.likeCount = result.likeCount;
      this.cdr.detectChanges();
    });
  }

  private loadResults(term: string, page: number) {
    this.postService.searchPosts(term, page, this.pageSize).subscribe((result) => {
      this.posts = result.items;
      this.currentPage = result.page;
      this.pageSize = result.pageSize;
      this.totalPages = result.totalPages;
      this.totalCount = result.totalCount;
      this.searched = true;
      this.cdr.detectChanges();
    });
  }
}
```

import { Component, OnDestroy, inject } from '@angular/core';
import { Router, RouterOutlet, RouterLink } from '@angular/router';
import { Subject, Subscription, debounceTime, distinctUntilChanged } from 'rxjs';

@Component({
  selector: 'app-root',
  standalone: true,
  imports: [RouterOutlet, RouterLink],
  template: `
    <nav class="navbar navbar-expand-lg navbar-light bg-light mb-4">
      <div class="container">
        <span class="navbar-brand">DevBlog</span>
        <div class="navbar-nav">
          <a class="nav-link" routerLink="/posts">Posts</a>
          <a class="nav-link" routerLink="/login">Login</a>
        </div>
        <input
          type="search"
          class="form-control ms-auto"
          style="max-width: 240px;"
          placeholder="Ara..."
          (input)="onSearchInput($event)" />
      </div>
    </nav>
    <main>
      <router-outlet />
    </main>
  `
})
export class AppComponent implements OnDestroy {
  private router = inject(Router);
  private searchTerms$ = new Subject<string>();
  private subscription: Subscription;

  constructor() {
    this.subscription = this.searchTerms$
      .pipe(debounceTime(300), distinctUntilChanged())
      .subscribe((term) => {
        if (term.trim()) {
          this.router.navigate(['/search'], { queryParams: { q: term, page: 1 } });
        }
      });
  }

  onSearchInput(event: Event) {
    const value = (event.target as HTMLInputElement).value;
    this.searchTerms$.next(value);
  }

  ngOnDestroy() {
    this.subscription.unsubscribe();
  }
}
```
